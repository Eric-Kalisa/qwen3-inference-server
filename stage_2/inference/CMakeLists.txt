cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 90)
endif()
project(qwen3_cuda LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
find_package(CUDAToolkit 13.3 REQUIRED)
add_subdirectory(third_party/tokenizer.cpp tokenizer_cpp)
set_target_properties(tokenizer_lib onig PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(qwen3_tokenizer SHARED tokenizer_c_api.cpp)
target_link_libraries(qwen3_tokenizer PRIVATE tokenizer_lib)
add_library(qwen3_runtime SHARED runtime.cu)
set_target_properties(qwen3_runtime PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
target_link_libraries(qwen3_runtime PRIVATE CUDA::cublas CUDA::cudart)
target_compile_options(qwen3_runtime PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--enable-tile;-lineinfo>")
